SCRIPT.JS

/**
 * AuraSmart Home - Core Logic
 * Senior Frontend Implementation
 */

// --- State Management ---
let state = {
    masterAcousticControl: true,
    devices: []
};

// Load data from localStorage on startup
function init() {
    const savedData = localStorage.getItem('auraSmartData');
    if (savedData) {
        state = JSON.parse(savedData);
    } else {
        // Default starting data for first-time users
        state = {
            masterAcousticControl: true,
            devices: [
                {
                    id: 'dev_' + Date.now(),
                    deviceName: 'Living Room Hub',
                    deviceUid: '8821-X',
                    online: true,
                    switches: [
                        { id: 'sw_1', name: 'Main Light', status: true, acousticEnabled: true },
                        { id: 'sw_2', name: 'AC Unit', status: false, acousticEnabled: false }
                    ]
                }
            ]
        };
        saveState();
    }
    render();
}

function saveState() {
    localStorage.setItem('auraSmartData', JSON.stringify(state));
}

// --- UI Rendering ---
function render() {
    const container = document.getElementById('dashboard-container');
    const masterToggle = document.getElementById('master-toggle');
    
    if (masterToggle) {
        masterToggle.checked = state.masterAcousticControl;
    }

    // Clear existing cards
    container.innerHTML = '';

    state.devices.forEach((device, dIdx) => {
        const card = document.createElement('section');
        card.className = 'device-card';
        
        // Building the exact layout required
        card.innerHTML = `
            <div class="card-header">
                <div class="device-info">
                    <span class="status-led ${device.online ? 'online' : 'offline'}"></span>
                    <span class="device-id">${device.deviceName} <small>ID: ${device.deviceUid}</small></span>
                </div>
                <button class="btn-remove" onclick="removeDevice(${dIdx})" title="Remove Device">
                    <i class="ph ph-trash"></i>
                </button>
            </div>

            <div class="switch-grid">
                ${device.switches.map((sw, sIdx) => `
                    <div class="switch-tile">
                        <div class="tile-top">
                            <label class="glass-switch">
                                <input type="checkbox" ${sw.status ? 'checked' : ''} 
                                       onchange="toggleSwitch(${dIdx}, ${sIdx})">
                                <span class="slider"></span>
                            </label>
                            
                            <div class="acoustic-group">
                                <i class="ph ph-microphone-polaroid mic-icon"></i>
                                <label class="glass-switch-mini">
                                    <input type="checkbox" ${sw.acousticEnabled ? 'checked' : ''} 
                                           onchange="toggleAcoustic(${dIdx}, ${sIdx})">
                                    <span class="slider"></span>
                                </label>
                            </div>
                        </div>
                        <div class="tile-bottom">
                            <span class="switch-name">${sw.name}</span>
                            <button class="btn-edit" onclick="renameSwitch(${dIdx}, ${sIdx})">
                                <i class="ph ph-pencil-simple"></i>
                            </button>
                        </div>
                    </div>
                `).join('')}
            </div>

            <div class="card-footer">
                <button class="btn-add-switch" onclick="addSwitch(${dIdx})">
                    <i class="ph ph-plus-circle"></i> Add Switch
                </button>
            </div>
        `;
        container.appendChild(card);
    });
}

// --- Action Functions ---
// Attached to 'window' so that HTML 'onclick' attributes can access them

window.toggleSwitch = function(dIdx, sIdx) {
    state.devices[dIdx].switches[sIdx].status = !state.devices[dIdx].switches[sIdx].status;
    saveState();
};

window.toggleAcoustic = function(dIdx, sIdx) {
    state.devices[dIdx].switches[sIdx].acousticEnabled = !state.devices[dIdx].switches[sIdx].acousticEnabled;
    saveState();
};

window.addSwitch = function(dIdx) {
    const switchName = prompt("Enter Switch Name (e.g., Ceiling Fan):");
    if (!switchName) return;

    state.devices[dIdx].switches.push({
        id: 'sw_' + Date.now(),
        name: switchName,
        status: false,
        acousticEnabled: true // New switches default to having acoustic control enabled
    });

    saveState();
    render();
};

window.renameSwitch = function(dIdx, sIdx) {
    const currentName = state.devices[dIdx].switches[sIdx].name;
    const newName = prompt("Rename Switch:", currentName);
    
    if (newName) {
        state.devices[dIdx].switches[sIdx].name = newName;
        saveState();
        render();
    }
};

window.removeDevice = function(dIdx) {
    if (confirm("Are you sure you want to remove this device and all its switches?")) {
        state.devices.splice(dIdx, 1);
        saveState();
        render();
    }
};

// --- Event Listeners for Page Load ---
document.addEventListener('DOMContentLoaded', () => {
    init();

    // Global Master Toggle Logic
    const masterToggle = document.getElementById('master-toggle');
    if (masterToggle) {
        masterToggle.addEventListener('change', (e) => {
            state.masterAcousticControl = e.target.checked;
            saveState();
        });
    }

    // FAB Add Device Logic
    const fabBtn = document.getElementById('fab-add-device');
    if (fabBtn) {
        fabBtn.addEventListener('click', () => {
            const name = prompt("Enter Device Name (e.g., Bedroom Hub):");
            if (!name) return;

            const uid = prompt("Enter Device ID (e.g., 1234-X):", "XXXX-X");
            
            const newDevice = {
                id: 'dev_' + Date.now(),
                deviceName: name,
                deviceUid: uid || 'XXXX-X',
                online: true,
                switches: []
            };

            state.devices.push(newDevice);
            saveState();
            render();
        });
    }
});
